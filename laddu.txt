5.Write a program in prolog to solve 4-Queens problem

% 4-Queens Problem
queens(Q) :-
permutation([1,2,3,4], Q),
    safe(Q).
safe([]).
safe([Q|Qs]) :-
safe(Qs),
no_attack(Q, Qs, 1).
no_attack(_, [], _).
no_attack(Q, [Q2|Qs], D) :-
    Q =\= Q2,
abs(Q - Q2) =\= D,
    D1 is D + 1,
no_attack(Q, Qs, D1).

Queries
?- queens(Q).

6.Write a program in prolog to solve traveling salesman problem

% Traveling Salesman Problem
road(a,b,10).
road(a,c,15).
road(a,d,20).
road(b,c,35).
road(b,d,25).
road(c,d,30).
distance(X,Y,D) :- road(X,Y,D).
distance(X,Y,D) :- road(Y,X,D).
tsp(Path,Cost) :-
    permutation([b,c,d],P),
    Path = [a|P],
append(Path,[a],Route),
cost(Route,Cost).
cost([_],0).
cost([A,B|T],Cost) :-
    distance(A,B,D),
    cost([B|T],C),
    Cost is D+C.

Query
?-tsp(Path, Cost).
