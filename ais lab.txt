% Ensure no queens attack diagonally 
safe([]). 
safe([Q|Others]) :- 
safe(Others), 
no_attack(Q, Others, 1). 
no_attack(_, [], _). 
no_attack(Q, [Q1|Rest], D) :- 
Q =\= Q1 + D, 
Q =\= Q1 - D, 
D1 is D + 1, 
no_attack(Q, Rest, D1). 
% Permutation helper 
permutation([], []). 
permutation(L, [H|T]) :- 
select(H, L, R), 
permutation(R, T).

% Main DFS predicate 
dfs(Start, Goal, Path) :- 
dfs_helper(Start, Goal, [Start], Path). 
% Base case: Goal reached 
dfs_helper(Goal, Goal, Visited, Path) :- 
reverse(Visited, Path). 
% Recursive case: explore next edge 
dfs_helper(Current, Goal, Visited, Path) :- 
edge(Current, Next), 
\+ member(Next, Visited), 
dfs_helper(Next, Goal, [Next|Visited], Path).
