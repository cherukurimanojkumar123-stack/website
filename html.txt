Program 8:      write java script to find factorial of a given number

<html>
<head>
<title>Calculating Factorial</title>
<script  language="javascript">

function factorial(n)
  {
	var f=1,i;
	for(i=1;i<=n;i++)
		f=f*i;
return f;
  }
</script>
</head>
<body>
<script language="javascript">
var n=parseInt(prompt("enter the number"," "));
var f=factorial(n);
document.write("factorial value="+f);
</script>
</body>
</html>

Program 9:      write java script to Calculate Sum of Digits of a Number

<html>
<head>
<title>Calculating Sum of Digits of a Number</title>
</head>
<body>
<script language="javascript">
var n=parseInt(prompt("enter the number"," "));
var sum=0,r;
	while(n!=0)
	{
		r=n%10;
		sum=sum+r;
		n=Math.floor(n/10);
	}
document.writeln("sum of digits="+sum);
</script>
</body>
